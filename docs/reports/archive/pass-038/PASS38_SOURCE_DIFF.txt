+pub(crate) fn encode_semantic_module_specs(
+    out: &mut Vec<u8>,
+    specs: &[BuiltinSemanticModuleSpec],

+    push_len(out, specs.len())?;

+    }
+    Ok(())

+pub(crate) fn decode_semantic_module_specs(
+    cursor: &mut Cursor<'_>,
+) -> Result<Vec<BuiltinSemanticModuleSpec>, &'static str> {
