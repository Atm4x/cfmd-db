+        let mut scan_ordinals = vec![None; leaf_count];

+    fn merge(&self, other: &Self) -> Option<Self> {
+        if self
+            .fragments
+            .iter()
+            .zip(&other.fragments)
+            .any(|(left, right)| left.is_some() && right.is_some())
+        {
+            return None;
+        }

+            fragments: self
+                .fragments
