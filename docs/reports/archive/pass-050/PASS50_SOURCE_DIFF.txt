+            else {

+            };

+                .get(*left_column)

+                .ok_or(RelQueryError::ColumnOutOfBounds)?;
+            let right_ref = right_columns
+                .get(*right_column)
+                .copied()
+                .ok_or(RelQueryError::ColumnOutOfBounds)?;
+            predicates.push(MultiwayJoinPredicate {
+                left: left_ref,
+                right: right_ref,
+                equivalence: *equivalence,
+            });
+            left_columns.extend(right_columns);
+            Ok(Some(left_columns))
+        }

+            left_column,
+            right_column,
+            equivalence,
