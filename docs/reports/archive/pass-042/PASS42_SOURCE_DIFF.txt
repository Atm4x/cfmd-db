+    #[must_use]
+    pub const fn choose_filter(row_count: usize, matching_rows: usize) -> SemanticAccessPath {
+        let scan_work = row_count;
+        let index_work = 1_usize.saturating_add(matching_rows);
+        if index_work < scan_work {
+            SemanticAccessPath::PersistedIndex
+        } else {
+            SemanticAccessPath::FullScan
+        }
+    }
+
+    #[must_use]
+    pub const fn choose_join(
+        left_rows: usize,
+        right_rows: usize,
+        right_distinct_keys: usize,
+    ) -> SemanticAccessPath {
+        if left_rows == 0 || right_rows == 0 || right_distinct_keys == 0 {
+            return SemanticAccessPath::PersistedIndex;
+        }
+        let nested_work = left_rows.saturating_mul(right_rows);
+        let average_bucket = right_rows.div_ceil(right_distinct_keys);
+        let indexed_work = left_rows.saturating_mul(1_usize.saturating_add(average_bucket));
+        if indexed_work < nested_work {
+            SemanticAccessPath::PersistedIndex
+        } else {
+            SemanticAccessPath::FullScan
+        }
+    }

+fn collect_direct_multi_key_join(plan: &Plan) -> Option<DirectMultiKeyJoin<'_>> {
+    let mut cursor = plan;
+    let mut extra_predicates = Vec::new();

+            second_column: *right_column,
+            equivalence: *equivalence,

+    }

+        right_column,

+    };

+        base_key: CompositeJoinKey {

+fn normalize_join_column_predicates(
+    join: &DirectMultiKeyJoin<'_>,
+    left_width: usize,
+    right_width: usize,
+) -> Option<Vec<CompositeJoinKey>> {
+    let total_width = left_width.checked_add(right_width)?;
+    let mut keys = Vec::with_capacity(join.extra_predicates.len().saturating_add(1));
+    keys.push(join.base_key);
+    for predicate in &join.extra_predicates {
+        if predicate.first_column >= total_width || predicate.second_column >= total_width {
+            return None;
+        }
+        let key = match (
+            predicate.first_column < left_width,
+            predicate.second_column < left_width,
+        ) {
+            (true, false) => CompositeJoinKey {
+                left_column: predicate.first_column,
+                right_column: predicate.second_column - left_width,
+                equivalence: predicate.equivalence,
+            },
+            (false, true) => CompositeJoinKey {
+                left_column: predicate.second_column,
+                right_column: predicate.first_column - left_width,
+                equivalence: predicate.equivalence,
+            },
+            (true, true) | (false, false) => return None,
+        };
+        if key.right_column >= right_width {
+            return None;
+        }
+        keys.push(key);
+    }

+fn binding_join_key_order(
+    binding: &SemanticIndexBinding,
+    keys: &[CompositeJoinKey],
+) -> Option<Vec<usize>> {
+    if binding.key_parts.len() != keys.len() {
+        return None;
+    }

+    let mut order = Vec::with_capacity(keys.len());

+        })?;

+fn rel_delta_filter_columns(
+    input_delta: RelationDelta,
+    left_column: usize,
+    right_column: usize,
+    equivalence: kernel_types::SemanticId,
+    context: &kernel_schema::SemanticContext,
+    registry: &kernel_semantics::SemanticRegistry,
+) -> Result<RelationDelta, RelQueryError> {
+    let left_type = input_delta
+        .result_type
+        .columns
+        .get(left_column)
+        .ok_or(RelQueryError::ColumnOutOfBounds)?;
+    let right_type = input_delta
+        .result_type
+        .columns
+        .get(right_column)
+        .ok_or(RelQueryError::ColumnOutOfBounds)?;
+    if !query_types_compatible(left_type, right_type, &context.schema) {
+        return Err(RelQueryError::TypeMismatch);
+    }
+    validate_query_equivalence(equivalence, left_type, context, registry)?;
+    validate_query_equivalence(equivalence, right_type, context, registry)?;
+    let left_input_equivalence =
+        relation_column_equivalence(&input_delta.result_type, left_column)?;
+    let right_input_equivalence =
+        relation_column_equivalence(&input_delta.result_type, right_column)?;
+    if !registry.equivalence_refines(context, left_input_equivalence, equivalence)?
+        || !registry.equivalence_refines(context, right_input_equivalence, equivalence)?
+    {
+        return Err(RelQueryError::EquivalenceNotCongruentWithInputEquality);
+    }
+
+    let filter_rows = |rows: Vec<Row>| -> Result<Vec<Row>, RelQueryError> {
+        rows.into_iter()
+            .filter_map(|row| {
+                let Some(left) = row.get(left_column) else {
+                    return Some(Err(RelQueryError::ColumnOutOfBounds));
+                };
+                let Some(right) = row.get(right_column) else {
+                    return Some(Err(RelQueryError::ColumnOutOfBounds));
+                };
+                match registry.equivalent(context, equivalence, left, right) {
+                    Ok(true) => Some(Ok(row)),
+                    Ok(false) => None,
+                    Err(error) => Some(Err(error.into())),
+                }
+            })
+            .collect()
+    };
+
+    Ok(RelationDelta {
+        inserted: filter_rows(input_delta.inserted)?,
+        removed: filter_rows(input_delta.removed)?,
+        result_type: input_delta.result_type,
+    })
