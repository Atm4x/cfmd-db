+impl InstalledRelation {
+    fn new(data: NativeRelation) -> Self {
+        let row_count = native_row_count(&data);
+        let mut row_ids = Vec::with_capacity(row_count);
+        let mut positions = Vec::with_capacity(row_count);
+        for index in 0..row_count {
+            row_ids.push(PhysicalRowId(index));
+            positions.push(Some(index));
+        }
+        Self {
+            data,
+            row_ids,
+            positions,
+            positions_are_identity: true,
+        }
+    }
+
+    fn row_id_at(&self, index: usize) -> Result<PhysicalRowId, PhysicalExecutionError> {
+        self.row_ids
+            .get(index)
+            .copied()
+            .ok_or(PhysicalExecutionError::ColumnShapeMismatch)
+    }
+
+    fn position(&self, id: PhysicalRowId) -> Option<usize> {
+        if self.positions_are_identity {
+            return (id.0 < self.row_ids.len()).then_some(id.0);
+        }
+        self.positions.get(id.0).copied().flatten()
+    }
+
+    fn remove_row(&mut self, index: usize) -> Result<PhysicalRowId, PhysicalExecutionError> {
+        let id = self.row_id_at(index)?;
+        remove_native_row(&mut self.data, index)?;
+        self.row_ids.remove(index);
+        self.positions_are_identity = false;
+        self.positions[id.0] = None;
+        for (position, row_id) in self.row_ids.iter().copied().enumerate().skip(index) {
+            self.positions[row_id.0] = Some(position);
+        }
+        Ok(id)
+    }
+
+    fn push_row(
+        &mut self,
+        row: &kernel_query::Row,
+    ) -> Result<PhysicalRowId, PhysicalExecutionError> {
+        push_native_row(&mut self.data, row)?;
+        let id = PhysicalRowId(self.positions.len());
+        let position = self.row_ids.len();
+        self.row_ids.push(id);
+        self.positions.push(Some(position));
+        Ok(id)
+    }
