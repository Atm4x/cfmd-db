+            let i64_binding = semantic_binding_as_i64(&binding, context, registry)?;
+            let has_i64 = i64_binding.is_some_and(|binding| store.i64_index(binding).is_some());
+            let has_semantic = match store.semantic_indexes.get(&binding) {
+                Some(state) => state.compatible_with(context, registry)?,
+                None => false,
+            };
+            if !has_i64 && !has_semantic {
+                let left_rows =
+                    native_row_count(&store.installed(left_relation, left_layout)?.data);
+                let right_rows =
+                    native_row_count(&store.installed(right_relation, right_layout)?.data);
+                if left_rows != 0 && right_rows != 0 {
+                    let scan_work = SemanticAccessCostModel::join_scan_work(left_rows, right_rows);
+                    let optimistic_ephemeral = SemanticAccessCostModel::ephemeral_join_work(
+                        left_rows,
+                        right_rows,
+                        right_rows,
+                        binding.key_parts.len(),
+                    );
+                    if optimistic_ephemeral < scan_work {
+                        uses.entry(binding)
+                            .or_default()
+                            .push(SemanticStatisticsAdviceUse {
+                                left_rows,
+                                right_rows,
+                                expected_executions,
+                            });
+                    }
+                }
+            }
+        }
+        return Ok(());
+    }
+    match plan {
+        Plan::FilterEqConst { input, .. }
+        | Plan::FilterEqColumns { input, .. }
+        | Plan::Project { input, .. }
+        | Plan::Distinct { input, .. }
+        | Plan::Group { input, .. }
+        | Plan::TopKWithTies { input, .. }
+        | Plan::PromoteToBag(input) => collect_semantic_statistics_advice_uses(
+            input,
+            store,
+            context,
+            registry,
+            expected_executions,
+            uses,
+        ),
+        Plan::JoinEq { left, right, .. } => {
+            collect_semantic_statistics_advice_uses(
+                left,
+                store,
+                context,
+                registry,
+                expected_executions,
+                uses,
+            )?;
+            collect_semantic_statistics_advice_uses(
+                right,
+                store,
+                context,
+                registry,
+                expected_executions,
+                uses,
+            )
+        }
+        Plan::Scan { .. } => Ok(()),
+    }

+fn semantic_statistics_savings(
+    uses: &[SemanticStatisticsAdviceUse],
+    distinct_key_count: usize,
+    key_parts: usize,
+) -> u128 {
+    uses.iter().fold(0_u128, |total, use_| {
+        let scan_work = SemanticAccessCostModel::join_scan_work(use_.left_rows, use_.right_rows);
+        let actual_ephemeral = SemanticAccessCostModel::ephemeral_join_work(
+            use_.left_rows,
+            use_.right_rows,
+            distinct_key_count.max(1),
+            key_parts,
+        );
+        let savings = actual_ephemeral.saturating_sub(scan_work) as u128;
+        total.saturating_add(savings.saturating_mul(use_.expected_executions as u128))
+    })

+fn semantic_statistics_advice_choice(
+    store: &PhysicalStore,
+    binding: SemanticIndexBinding,
+    uses: &[SemanticStatisticsAdviceUse],
+    context: &kernel_schema::SemanticContext,
+    registry: &kernel_semantics::SemanticRegistry,
+    report: &mut SemanticStatisticsAdvisorReport,
+) -> Result<Option<SemanticStatisticsAdviceChoice>, PhysicalExecutionError> {
+    let relation = store.installed(binding.relation, binding.layout)?;
+    let row_count = native_row_count(&relation.data);
+    let existing = store.semantic_statistics.get(&binding);
+    let compatible_existing = match existing {
+        Some(state) => state.compatible_with(context, registry)? && state.row_count == row_count,
+        None => false,
+    };
+    let build_work = if compatible_existing {
+        0
+    } else {
+        row_count.saturating_mul(binding.key_parts.len().max(1)) as u128
+    };
+    if !compatible_existing {
+        let upper_bound = semantic_statistics_savings(uses, 1, binding.key_parts.len());
+        if upper_bound <= build_work {
+            report.rejected_unprofitable.push(binding);
+            return Ok(None);
+        }
+    }
+    let (distinct_key_count, estimated_bytes, prepared_state) = if compatible_existing {
+        let state = existing.expect("compatible statistics must exist");
+        (
+            state.counts.len(),
+            semantic_statistics_estimated_retained_bytes(state),
+            None,
+        )
+    } else {
+        let candidate = MaterializedSemanticStatisticsState::build(
+            binding.clone(),
+            relation,
+            context,
+            registry,
+        )?;
+        let estimated_bytes = semantic_statistics_estimated_retained_bytes(&candidate);
+        (candidate.counts.len(), estimated_bytes, Some(candidate))
+    };
+    let gross_savings =
+        semantic_statistics_savings(uses, distinct_key_count, binding.key_parts.len());
+    if gross_savings <= build_work {
+        report.rejected_unprofitable.push(binding);
+        return Ok(None);
+    }
+    let advisor_managed =
+        store
+            .advisor_managed_artifacts
+            .contains(&ManagedPhysicalArtifact::SemanticStatistics(
+                binding.clone(),
+            ));
+    let replaced_fixed_bytes = if !compatible_existing && !advisor_managed {
+        existing.map_or(0, |state| {
+            semantic_statistics_estimated_retained_bytes(state)
+        })
+    } else {
+        0
+    };
+    Ok(Some(SemanticStatisticsAdviceChoice {
+        binding,
+        net_savings: gross_savings.saturating_sub(build_work),
+        row_count,
+        estimated_bytes,
+        compatible_existing,
+        advisor_managed,
+        replaced_fixed_bytes,
+        prepared_state,
+    }))
+}

+fn semantic_statistics_advice_choices(
+    store: &PhysicalStore,
+    workload: &[SemanticIndexWorkloadSample],
+    context: &kernel_schema::SemanticContext,
+    registry: &kernel_semantics::SemanticRegistry,
+    report: &mut SemanticStatisticsAdvisorReport,
+) -> Result<Vec<SemanticStatisticsAdviceChoice>, PhysicalExecutionError> {
+    let mut uses = BTreeMap::<SemanticIndexBinding, Vec<SemanticStatisticsAdviceUse>>::new();
+    for sample in workload {
+        collect_semantic_statistics_advice_uses(
+            &sample.plan,
+            store,
+            context,
+            registry,
+            sample.expected_executions,
+            &mut uses,
+        )?;
+    }
+    let mut choices = Vec::new();
+    for (binding, uses) in uses {
+        if let Some(choice) =
+            semantic_statistics_advice_choice(store, binding, &uses, context, registry, report)?
+        {
+            choices.push(choice);
+        }
+    }
+    choices.sort_by(|left, right| {
+        let left_cost = if left.compatible_existing && !left.advisor_managed {
+            0
+        } else {
+            left.estimated_bytes
+        };
+        let right_cost = if right.compatible_existing && !right.advisor_managed {
+            0
+        } else {
+            right.estimated_bytes
+        };
+        match (left_cost, right_cost) {
+            (0, 0) => right
+                .net_savings
+                .cmp(&left.net_savings)
+                .then_with(|| left.binding.cmp(&right.binding)),
+            (0, _) => std::cmp::Ordering::Less,
+            (_, 0) => std::cmp::Ordering::Greater,
+            _ => {
+                let left_density = left.net_savings.saturating_mul(right_cost as u128);
+                let right_density = right.net_savings.saturating_mul(left_cost as u128);
+                right_density
+                    .cmp(&left_density)
+                    .then_with(|| right.net_savings.cmp(&left.net_savings))
+                    .then_with(|| right.row_count.cmp(&left.row_count))
+                    .then_with(|| left.binding.cmp(&right.binding))
+            }
+        }
+    });
+    Ok(choices)
+}

+fn semantic_statistics_advisor_selection(
+    store: &PhysicalStore,
+    workload: &[SemanticIndexWorkloadSample],
+    policy: PhysicalArtifactAdvisorPolicy,
+    context: &kernel_schema::SemanticContext,
+    registry: &kernel_semantics::SemanticRegistry,
+) -> Result<SemanticStatisticsAdvisorSelection, PhysicalExecutionError> {
+    let mut report = SemanticStatisticsAdvisorReport::default();
+    let choices =
+        semantic_statistics_advice_choices(store, workload, context, registry, &mut report)?;
+    let current_memory = store
+        .artifact_memory_report()
+        .total_estimated_retained_bytes;
+    let replaceable_managed_bytes = saturating_usize_sum(
+        store
+            .semantic_statistics
+            .iter()
+            .filter(|(binding, _)| {
+                store.advisor_managed_artifacts.contains(
+                    &ManagedPhysicalArtifact::SemanticStatistics((*binding).clone()),
+                )
+            })
+            .map(|(_, state)| semantic_statistics_estimated_retained_bytes(state)),
+    );
+    let mut fixed_estimated_bytes = current_memory.saturating_sub(replaceable_managed_bytes);
+    let mut selected = BTreeSet::new();
+    let mut prepared_states = Vec::new();
+    let mut managed_estimated_bytes = 0_usize;
+    for mut choice in choices {
+        let managed_cost = if choice.compatible_existing && !choice.advisor_managed {
+            0
+        } else {
+            choice.estimated_bytes
+        };
+        let projected_fixed_estimated_bytes =
+            fixed_estimated_bytes.saturating_sub(choice.replaced_fixed_bytes);
+        if managed_estimated_bytes.saturating_add(managed_cost) > policy.max_managed_estimated_bytes
+            || projected_fixed_estimated_bytes
+                .saturating_add(managed_estimated_bytes)
+                .saturating_add(managed_cost)
+                > policy.max_total_estimated_bytes
+        {
+            report.rejected_budget.push(choice.binding);
+            continue;
+        }
+        managed_estimated_bytes = managed_estimated_bytes.saturating_add(managed_cost);
+        fixed_estimated_bytes = projected_fixed_estimated_bytes;
+        if let Some(state) = choice.prepared_state.take() {
+            prepared_states.push((choice.binding.clone(), state));
+        }
+        selected.insert(choice.binding);
+    }
+    report.fixed_estimated_bytes = fixed_estimated_bytes;
+    report.total_estimated_bytes_after =
+        fixed_estimated_bytes.saturating_add(managed_estimated_bytes);
+    Ok(SemanticStatisticsAdvisorSelection {
+        selected,
+        prepared_states,
+        managed_estimated_bytes,
+        report,
+    })
+}
