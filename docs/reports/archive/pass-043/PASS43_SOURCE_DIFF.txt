+    workload: &[SemanticIndexWorkloadSample],

+                observation

+                    .saturating_mul(sample.expected_executions as u128),

+fn semantic_index_advice_choices(
+    store: &PhysicalStore,
+    scores: BTreeMap<SemanticIndexBinding, SemanticIndexAdviceScore>,
+    context: &kernel_schema::SemanticContext,
+    registry: &kernel_semantics::SemanticRegistry,
+    report: &mut SemanticIndexAdvisorReport,
+) -> Result<Vec<SemanticIndexAdviceChoice>, PhysicalExecutionError> {
+    let mut choices = Vec::new();
+    for score in scores.into_values() {
+        let compatible_existing = match store.semantic_indexes.get(&score.binding) {
+            Some(state) => state.compatible_with(context, registry)?,
+            None => false,
+        };
+        let build_work = if compatible_existing {
+            0
+        } else {
+            score.key_cells as u128
+        };
+        if score.gross_savings <= build_work {
+            report.rejected_unprofitable.push(score.binding);
+            continue;
+        }
+        let advisor_managed = store
+            .advisor_managed_semantic_indexes
+            .contains(&score.binding);
+        choices.push(SemanticIndexAdviceChoice {
+            binding: score.binding,
+            net_savings: score.gross_savings - build_work,
+            key_cells: score.key_cells,
+            compatible_existing,
+            advisor_managed,
+        });
+    }
+    Ok(choices)

+fn compare_semantic_index_advice_choice(
+    left: &SemanticIndexAdviceChoice,
+    right: &SemanticIndexAdviceChoice,
+) -> std::cmp::Ordering {
+    let left_cost = semantic_index_managed_cost(left);
+    let right_cost = semantic_index_managed_cost(right);
+    match (left_cost, right_cost) {
+        (0, 0) => right
+            .net_savings
+            .cmp(&left.net_savings)
+            .then_with(|| left.binding.cmp(&right.binding)),
+        (0, _) => std::cmp::Ordering::Less,
+        (_, 0) => std::cmp::Ordering::Greater,
+        _ => {
+            let left_density = left.net_savings.saturating_mul(right_cost as u128);
+            let right_density = right.net_savings.saturating_mul(left_cost as u128);
+            right_density
+                .cmp(&left_density)
+                .then_with(|| right.net_savings.cmp(&left.net_savings))
+                .then_with(|| left.binding.cmp(&right.binding))
+        }
+    }

+fn semantic_index_advisor_selection(
+    store: &PhysicalStore,
+    workload: &[SemanticIndexWorkloadSample],
+    policy: SemanticIndexAdvisorPolicy,
+    context: &kernel_schema::SemanticContext,
+    registry: &kernel_semantics::SemanticRegistry,

+    let mut report = SemanticIndexAdvisorReport::default();

+        }
+        managed_key_cells = managed_key_cells.saturating_add(managed_cost);

+    }
