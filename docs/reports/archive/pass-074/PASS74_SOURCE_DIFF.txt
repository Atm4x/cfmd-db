+fn transpose_value_rows(
+    rows: &[kernel_query::Row],
+    column_count: usize,
+) -> Result<Vec<Vec<Value>>, PhysicalExecutionError> {
+    let mut columns = (0..column_count)
+        .map(|_| Vec::with_capacity(rows.len()))
+        .collect::<Vec<_>>();
+    for row in rows {
+        if row.len() != column_count {

+        }

+    }
